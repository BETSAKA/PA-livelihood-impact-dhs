2  Treatment assignation

Les clusters d’enquêtes DHS (1997, 2008 et 2021), MIS (2011, 2013, 2016) et MICS (2018) sont classés par rapport à leur distance aux aires protégées (AP) et à leur localisation en zone rurale ou urbaine. Le résultat attendu est le suivant :

  • Groupe de traitement: Ménages localisées dans les clusters ruraux à moins de 10 km d’une AP de l’univers de traitement (créée à partir de 2008 pendant la période d’étude) — appartenance au groupe, y compris avant l’entrée en vigueur de l’AP

  • Groupe de contrôle: Ménages localisées dans les clusters ruraux à plus de 10 km des AP de l’univers de traitement

  • Groupe à exclure: Ménages localisées dans les clusters à moins de 10 km des AP créées avant 2008 et dans la zone urbaine

2.3 Une base actualisée des limites des aires protégées

La base dynamic_wdpa.gpkg est le produit consolidé du travail intitulé « Dynamic conservation boundaries » (Bédécarrats et al. 2026), dont le code source est archivé sur GitHub et les données publiées sur un dépôt de données scientifique. Elle répond à plusieurs limite structurelle du World Database on Protected Areas (WDPA), qui ne conserve que l’état actuel de chaque site : à chaque modification (redécoupage, reclassement, dégazage, protection temporaire), la configuration antérieure est écrasée, ce qui rend impossibles les analyses rétrospectives et biaise les évaluations d’impact qui assimilent les limites actuelles à celles en vigueur par le passé. En outre les chercheurs utilisent couramment le champ STATUS_YR comme la date de création des aires protégées, mais ce n’est pas toujours le cas, car cette date est réinitialisée à chaque changement de statut ou de dénomination d’une aire protégée. Pour remédier à ces faille, la base dynamic_wdpa.gpkg documente 123 aires protégées terrestres de Madagascar de 2000 à 2025 sous forme d’états temporels bornés dans le temps : chaque ligne représente une zone dans un état donné, avec sa période de validité (valid_from/valid_to), sa géométrie (WGS84), et ses attributs (WDPAID, nom, désignation, catégorie UICN, statut, autorité de gestion), chaque état étant rattaché à une source juridique (registre d’amendements YAML consolidé par un moteur déterministe en R). Les zones sont typées par le champ zone_type : external_boundary (limite externe de l’AP), core_zone (noyau dur où l’usage extractif est prohibé), internal_buffer_zone et external_buffer_zone (zones tampons internes et externes). C’est ce découpage temporel qui permet, dans ce chapitre, de classer chaque cluster d’enquête selon la configuration des AP en vigueur à sa date d’enquête (cf. section « Classification des clusters »), et de mener le test de sensibilité « limites fixes vs limites à date ».

# Fonction générique (destinée à conservation-dynamic-madagascar) : réseau d'AP
# en vigueur à une date de référence de `year`. La date de référence coupe
# l'année en deux : les enquêtes DHS/MICS s'étalent sur plusieurs mois, le
# 1er juin approche mieux la date moyenne d'interview que le 31 décembre,
# qui post-date la collecte de presque toute l'année.
#' Get the PA network in force at a reference date
#'
#' Filters consolidated PA states to those valid at the reference date of a
#' given year: states whose validity started on or before the cut-off date and
#' have not ended by then (open-ended states are always kept). The cut-off
#' date is expressed as "DD/MM" (day/month) and defaults to "01/06"
#' (June 1st), which better approximates the average DHS/MICS interview date
#' than December 31st.
#'
#' @param states Consolidated PA states (output from consolidate_pa_states)
#' @param year Reference year
#' @param cutoff_date Cut-off date within the year, as "DD/MM" (default "01/06")
#' @return sf object with the states valid at the reference date
get_pa_network <- function(states, year, cutoff_date = "01/06") {
  md <- as.integer(strsplit(cutoff_date, "/")[[1]])
  ref_date <- as.Date(sprintf("%d-%02d-%02d", year, md[2], md[1]))

  states |>
    filter(valid_from <= ref_date, is.na(valid_to) | ref_date < valid_to)
}

# Date de référence des vagues : un seul point de définition, réutilisé
# partout (classification, tests, estimation). Le 1er juin coupe l'année en
# deux et approche la date moyenne d'interview DHS/MICS.
survey_reference_date <- function(year) {
  as.Date(sprintf("%d-06-01", year))
}

# Limites effectives d'un ensemble d'AP à une date de référence :
# - AP future à cette date (treatment_date > ref) : limite en vigueur à
#   l'entrée en traitement (premier état) ;
# - AP déjà entrée en traitement : limite en vigueur à la date de référence ;
#   si aucun état ne couvre cette date (AP dissoute temporairement ou
#   définitivement), on retient le dernier état antérieur — repli documenté,
#   la définition du traitement du PAP reposant sur la création, non sur la
#   dissolution.
pa_effective_states <- function(ids, ref_date, fixed = FALSE) {
  if (fixed) {
    # Test de sensibilité : dernier état de limite connu, appliqué tel quel.
    return(pa_states_fixed |> filter(WDPAID %in% ids))
  }
  pa_states_analysis |>
    filter(WDPAID %in% ids) |>
    left_join(pa_first |> select(WDPAID, treatment_date), by = "WDPAID") |>
    filter(valid_from <= pmax(ref_date, treatment_date)) |>
    group_by(WDPAID) |>
    slice_max(valid_from, n = 1, with_ties = FALSE) |>
    ungroup() |>
    select(-treatment_date)
}

# Couches d'AP d'une vague : pré-baseline (Excluded) et univers de traitement
# (GROUP = Treatment), avec la distinction actif / futur à la date de référence.
# Exception placebo : la vague 1997 utilise l'univers et les limites de 2008,
# de sorte que sa sélection de clusters est identique à celle de 2008.
pa_layers <- function(year, fixed = FALSE) {
  ref_year <- max(year, 2008)
  ref_date <- survey_reference_date(ref_year)
  if (fixed) {
    pre  <- pa_states_fixed |> filter(WDPAID %in% pa_pre_ids)
    univ <- pa_states_fixed |> filter(WDPAID %in% pa_universe_ids)
  } else {
    net  <- get_pa_network(pa_states_analysis, ref_year)
    pre  <- net |> filter(WDPAID %in% pa_pre_ids)
    univ <- pa_effective_states(pa_universe_ids, ref_date)
  }
  univ <- univ |>
    left_join(pa_first |> select(WDPAID, treatment_date), by = "WDPAID") |>
    select(WDPAID, STATUS, STATUS_YR, IUCN_CAT, valid_from, valid_to,
           treatment_date, geometry)
  list(
    pre = pre,
    univ = univ,
    univ_active = univ |> filter(treatment_date <= ref_date),
    univ_future = univ |> filter(treatment_date >  ref_date)
  )
}

2.4 Création de buffer pour les AP avant et après 2008

On travaille toujours avec les portions terrestres des AP.

Pour analyser l’impact des AP selon leur période de création, on spécifie, pour chaque vague d’enquête, les AP antérieures à 2008 (en vigueur à la date de référence de la vague) et l’univers de traitement (AP créées à partir de 2008, limites effectives définies plus haut : limite de création pour les AP futures, limite en vigueur pour les AP actives). On applique ensuite un buffer zone de 10 km autour de chaque groupe d’AP. Chaque groupe a été reprojeté dans le système de projection officiel de Madagascar, qui est le Laborde (EPSG : 29702) pour les calculs de distances. Nous créons ensuite une carte pour visualiser les AP et leurs zones d’influence, en prenant la vague 2008 comme exemple. Exception : pour l’enquête 1997, utilisée comme test de placebo, on applique l’état du réseau d’AP de 2008 (et donc les buffers de 2008).

# Buffer de 10 km
buffer_dist <- 10000 

# Fonction de création du buffer de 10 km (reprojection Laborde pour les
# calculs de distance, union des AP, puis retour en WGS 84). Définie une
# seule fois ici et réutilisée par la classification des clusters.
buffer_10km <- function(pa, buffer_dist = 10000) {
  if (nrow(pa) == 0) return(st_as_sf(st_sfc(st_polygon(), crs = standard_crs)))
  pa |>
    st_transform(mdg_crs) |>
    st_buffer(dist = buffer_dist) |>
    st_make_valid() |>
    st_union() |>
    st_as_sf() |>
    st_make_valid() |>
    st_transform(standard_crs)
}

2.5 Classification des clusters

Les grappes d’enquête sont classées selon leur proximité aux AP créées avant 2008 et à partir de 2008. A l’aide de la fonction classify_clusters_with_pa(), nous catégorisons chaque cluster en trois groupes : traitement, contrôle ou exclu, selon leur localisation par rapport aux AP et leur statut urbain ou rural. Cette classification est appliquée aux clusters de 1997, 2008, 2011, 2013, 2016, 2018 et 2021. Parmi les clusters considérés comme “traités”, on identifie la date de début de traitement de l’AP, son année de création, son type et le numéro WDPAID ou l’identifiant unique de l’AP en question. Les résultats obtenus sont fusionnés. Nous créons ensuite une carte par année d’étude pour visualiser la localisation des AP et la répartition spatiale des clusters selon leur groupe.

Sémantique des variables de classification. Deux concepts distincts sont explicitement séparés :

  • GROUP (Treatment / Control / Excluded) désigne l’appartenance à la géographie de traitement éventuelle pendant la période d’étude : un cluster rural à moins de 10 km d’une AP de l’univers de traitement (créée entre 2008 et 2021, hors “Proposed”) est Treatment, y compris dans les vagues antérieures à la date de création de l’AP. C’est cette variable qui définit le groupe de traitement dans le DiD en double différence (l’activation du traitement est représentée par l’interaction avec la période post).
  • treated_now indique si l’AP est effectivement en vigueur à la date de référence de la vague (1er juin) : un cluster Treatment observé avant sa date de début de traitement a treated_now = 0, après cette date treated_now = 1.
  • treatment_date est la date de début de traitement de l’AP assignée (premier valid_from de la base dynamique, qui coïncide avec l’année STATUS_YR pour toutes les AP de l’univers, à l’exception correctif documenté du WDPAID 555697871) ; treatment_year reprend l’année de création STATUS_YR (définition substantielle inchangée).

Ce choix suit le plan de pré-analyse : le groupe de traitement du placebo 1997 est défini par la proximité des AP créées pendant la période de traitement ultérieure (2008–2021). On ne requalifie donc pas en Control une localisation traitée à terme au seul motif que l’AP n’existe pas encore à la date de l’enquête. En particulier, une AP créée tardivement dans l’année d’une vague (p. ex. oct.–déc. 2008, après le 1er juin) reste Treatment avec treated_now = 0 à la baseline 2008 — la comparaison STATUS_YR > année, qui perdait ces cas, est abandonnée au profit des dates.

Les limites des AP ayant évolué au cours du temps (le fichier dynamic_wdpa.gpkg documente chaque état des limites avec ses périodes de validité valid_from/valid_to), la classification est faite par vague d’enquête : pour chaque année, on utilise l’état du réseau d’AP valide à la date de référence (1er juin — la collecte s’étalant sur plusieurs mois, cette date coupe l’année en deux et approche la date moyenne d’interview), et non les limites actuelles. Pour une AP encore future à cette date, c’est la limite en vigueur à son entrée en traitement qui sert à identifier les clusters qui seront traités. La vague 1997 fait exception : elle sert de vague placebo (comparaison 1997–2008), on lui applique donc l’univers et les limites de 2008, de sorte qu’elle est construite avec le même univers de traitement et les mêmes règles de classification que la baseline 2008 (toutes les unités y sont pré-traitement : treated_now = 0). Il ne s’agit pas des mêmes unités échantillonnées : les vagues DHS/MIS sont des coupes répétées indépendantes et les identifiants de grappes ne sont pas des identifiants longitudinaux.

Les AP encore au statut “Proposed” (jamais officialisées pendant la période d’étude) sont exclues du jeu d’analyse : les clusters proches de ces AP sont classés Control, conformément au pré-analyse plan (aucune restriction n’existait pendant les enquêtes). Les clusters exposés à une AP antérieure à 2008 restent Excluded, y compris lorsque la zone chevauche celle d’une AP post-2008 (PAP : la date de création la plus ancienne l’emporte).

pa_buffered <- function(pa, buffer_dist = 10000) {
  pa |>
    select(WDPAID, STATUS_YR, IUCN_CAT, treatment_date, geometry) |>
    mutate(pa_geom = geometry) |>
    st_transform(mdg_crs) |>
    mutate(geometry = st_buffer(geometry, buffer_dist)) |>
    st_transform(standard_crs)
}

classify_clusters_with_pa <- function(cluster_gps, year, fixed = FALSE,
                                      buffer_dist = 10000) {
  # Couches effectives de la vague (1997 placebo -> univers et limites de 2008)
  lay <- pa_layers(year, fixed)
  ref_wave <- survey_reference_date(year)

  buf_pre  <- buffer_10km(lay$pre,  buffer_dist)
  buf_univ <- buffer_10km(lay$univ, buffer_dist)

  in_pre  <- st_within(cluster_gps, buf_pre,  sparse = FALSE)[, 1]
  in_univ <- st_within(cluster_gps, buf_univ, sparse = FALSE)[, 1]

  # GROUP = appartenance à la géographie de traitement éventuelle (pas
  # l'activation à la date d'enquête) ; les clusters exposés à une AP
  # antérieure à 2008 sont exclus, y compris en cas de chevauchement.
  base <- cluster_gps |>
    mutate(GROUP = case_when(
      URBAN_RURA == "U" ~ "Excluded",
      in_pre  ~ "Excluded",
      URBAN_RURA == "R" & in_univ ~ "Treatment",
      TRUE ~ "Control"
    ))

  # enrich treated with the earliest-treated + nearest PA
  treated_pts <- base |> filter(GROUP == "Treatment")
  if (nrow(treated_pts) == 0) {
    return(base |>
      mutate(WDPAID = NA_integer_, STATUS_YR = NA_integer_,
             IUCN_CAT = NA_character_, treatment_date = as.Date(NA),
             dist_km = NA_real_))
  }

  cand_pa <- pa_buffered(lay$univ, buffer_dist)
  cand <- st_join(treated_pts, cand_pa, join = st_within, left = FALSE) |>
    mutate(dist_km = as.numeric(st_distance(geometry, pa_geom, by_element = TRUE)) / 1000)

  best <- cand |>
    st_drop_geometry() |>
    group_by(DHSYEAR, DHSCLUST) |>
    slice_min(treatment_date, with_ties = TRUE) |>
    slice_min(dist_km, with_ties = FALSE) |>
    ungroup() |>
    select(DHSYEAR, DHSCLUST, WDPAID, STATUS_YR, IUCN_CAT, treatment_date, dist_km)

  base |>
    left_join(best, by = c("DHSYEAR", "DHSCLUST")) |>
    mutate(treated_now = as.integer(!is.na(treatment_date) & ref_wave >= treatment_date))
}

gps_all_class <- bind_rows(
  purrr::map(survey_years, function(y) {
    classify_clusters_with_pa(get(paste0("gps_", y)), y)
  })
)

# Function to read DHS data for the specified year and identifier
load_dhs_data <- function(dhs_folder, year, identifier) {
  folder_pattern <- paste0(".*", year, ".*", identifier)
  
  matching_folder <- list.dirs(dhs_folder, full.names = TRUE, recursive = TRUE) %>%
    keep(~ str_detect(.x, folder_pattern))
  
  if (length(matching_folder) == 0) {
    stop("No folder found for the specified year and identifier.")
  }
  
  if (identifier == "GE") {
    file_pattern <- "\\.shp$"
    data_loader <- function(file) st_read(file, quiet = TRUE)
  } else {
    file_pattern <- "\\.[Dd][Tt][Aa]$"
    data_loader <- read_dta
  }
  
  target_file <- list.files(matching_folder, pattern = file_pattern, full.names = TRUE)
  
  if (length(target_file) == 0) {
    stop("No valid file found in the folder.")
  }
  
  data <- data_loader(target_file)
  
  return(data)
}
